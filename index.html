function eliminarRegistro(index) { const pass = prompt("Ingrese la contraseña de seguridad para eliminar:"); if (pass === "septiembre10") { const regAEliminar = registros[index]; // Quitar de la pantalla inmediatamente registros.splice(index, 1); localStorage.setItem('power_epec_registros', JSON.stringify(registros)); renderizarTabla(); // Enviar orden de eliminación exacta a Google Sheets const scriptUrl = obtenerUrlScript(); if (scriptUrl && regAEliminar) { fetch(scriptUrl, { method: 'POST', mode: 'no-cors', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: "delete", rowIndex: regAEliminar.rowIndex, hora: regAEliminar.hora, equipo: regAEliminar.equipo }) }).then(() => { // Re-sincronizar tras 1.5 segundos para confirmar borrado de la hoja setTimeout(() => sincronizarDatos(false), 1500); }); } } else if (pass !== null) { alert("Contraseña incorrecta."); } }
